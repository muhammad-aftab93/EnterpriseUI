```csharp
public static class GeneratedGridFluent
{
    public static EnterpriseDataGrid<GridDemoRecord> Create(IEnumerable<GridDemoRecord> data) =>
        GridBuilder<GridDemoRecord>.Create()
            .Label("Sales order workbench")
            .Column("Id", "Record ID", 90d, c => { c.ReadOnly = true; c.PinPosition = ColumnPinPosition.Left; })
            .Column("Customer", "Enterprise Customer", 180d)
            .Column("Product", "Product Line", 170d)
            .Column("Region", "Region", 110d)
            .Column("UnitPrice", "Unit Price", 100d, c => { c.Alignment = ColumnTextAlignment.Right; c.FormatString = "C2"; c.ValidationRules.Add(new RangeRule(1m, 100000m) { Message = "Unit price must be positive" }); })
            .Column("Quantity", "Qty", 70d, c => { c.Alignment = ColumnTextAlignment.Right; c.FooterSummaries.Add(new GridColumnSummary(AggregateFunction.Average, "0.0")); c.ValidationRules.Add(new RangeRule(1, 999) { Message = "Quantity must be 1 to 999" }); })
            .Column("Total", "Gross Amount", 110d, c => { c.ReadOnly = true; c.Alignment = ColumnTextAlignment.Right; c.FormatString = "C2"; c.FooterSummaries.Add(new GridColumnSummary(AggregateFunction.Sum, "C0")); })
            .Column("Priority", "Priority", 90d, c => { c.ComboItems = new object[] { "Low", "Normal", "High", "Critical" }; })
            .Column("Status", "Status", 100d)
            .Column("Timestamp", "Date", 100d, c => { c.FormatString = "d"; })
            .Band("Order", "Id", "Customer", "Product", "Region")
            .Band("Pricing", "UnitPrice", "Quantity", "Total")
            .Band("Workflow", "Priority", "Status", "Timestamp")
            .Paging(GridPagingMode.Client)
            .Selection(GridSelectionMode.CheckboxRow)
            .Editing(GridEditMode.Batch)
            .FormatRule(new GridFormatRule { Name = "Gross bar", ColumnField = "Total", RuleType = FormatRuleType.DataBar, Value1 = 0m, Value2 = 60000m, BackgroundColorHex = "#5088d7" })
            .FormatRule(new GridFormatRule { Name = "Critical", ColumnField = "Priority", Operator = FormatConditionOperator.Equal, Value1 = "Critical", BackgroundColorHex = "#fff0c2", ForegroundColorHex = "#713f12" })
            .GroupSummary("Total", AggregateFunction.Sum, "Sum {0:C0}")
            .Configure(g =>
                {
                    g.ItemHeight = 28d;
                    g.ShowFilterRow = true;
                })
            .DataSource(data)
            .Build();
}

```

## Best practices

Validate on the column, edit in batch mode for review-and-save workflows, persist layouts per user, and export from the grid so the file matches what the user sees.

## Performance notes

Twenty thousand rows with paging, grouping, summaries and conditional formatting stay responsive because only the current page is rendered and queries run in the engine.

Measured numbers are in [PerformanceReport](Reports/PerformanceReport.md).

## Accessibility notes

Keyboard operable end to end (navigate, edit, save with Ctrl+S, search with Ctrl+F), announcements for query changes, errors in text, theme-derived colors.

## Troubleshooting

If Save changes reports errors, fix the cells marked with the error corner (hover for the message) or use Discard; validation blocks the commit by design.
